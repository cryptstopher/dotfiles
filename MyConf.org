** Text Manipulation

#+begin_src emacs-lisp
(defun csm/wrap-region (start end)
  "Prompt for a string and wrap the selected region with it."
  (interactive "r")
  (let ((input (read-string "Enter the surrounding character or string: ")))
    (save-excursion
      (goto-char end)
      (insert input)
      (goto-char start)
      (insert input))))

(defun insert-nice-date ()
  "Insert the current date in the format DD Month YYYY into the current buffer."
  (interactive)
  (insert (format-time-string "%d %B %Y")))

** Quote and Line Break Fixes

#+begin_src emacs-lisp
(defun fix-smart-quotes-and-carriage-returns ()
  "Replace smart quotes with straight quotes and add extra newline at each carriage return."
  (interactive)
  (goto-char (point-min))
  (while (re-search-forward "[""]" nil t)
    (replace-match "\"" nil nil))
  (goto-char (point-min))
  (while (re-search-forward "['']" nil t)
    (replace-match "'" nil nil))
  (goto-char (point-min))
  (while (search-forward "\r" nil t)
    (replace-match "\n\n" nil nil)))

(defun double-all-line-breaks ()
  "Make every line break into two line breaks, including carriage returns, newlines, and CRLF."
  (interactive)
  (goto-char (point-min))
  (while (search-forward "\r\n" nil t)
    (replace-match "\n\n" nil nil))
  (goto-char (point-min))
  (while (search-forward "\r" nil t)
    (replace-match "\n\n" nil nil))
  (goto-char (point-min))
  (while (search-forward "\n" nil t)
    (replace-match "\n\n" nil nil)))
#+end_src
